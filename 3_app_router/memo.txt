- app이라는 폴더 기반으로 라우팅이 결정
- 폴더 => page.tsx + layout.tsx로 구성
  - layout.tsx는 해당하는 하단 route전체에 적용됨
  - layout.tsx는 page.tsx를 감싼다.
  - !! 그 밑의 폴더에 layout.tsx추가시 중첩된다 !!
- params와 쿼리스트링이 페이지 컴포넌트의 props로 전달됨

------------------------------------------------------------------------

라우트그룹
- 경로상에는 영향을 미치지 않는 폴더
- 각기 다른 경로를 가지는 페이지 파일을 묶는다
- ()로 감싼 폴더 만들어서 정의

------------------------------------------------------------------------

export default async function SearchPage({
  searchParams,
}: {
  searchParams: Promise<{ q: string }>;
}) {
  const { q } = await searchParams;

  return <div>검색 페이지 q: {q}</div>;
}

------------------------------------------------------------------------

export default async function BookDetailPage({
  params,
}: {
  params: Promise<{ id: string }>;
}) {
  const { id } = await params;

  return <div>book/[id] {id} </div>;
}
